# File > New starts from template.json, compiled in so the two cannot drift.
# Editing the template reconfigures. One raw string literal, so the text may
# not close its delimiter or carry an @ for file(CONFIGURE), and MSVC caps a
# literal at 16380 bytes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS template.json)
file(READ template.json COLOURWARS_NEW_LEVEL_TEMPLATE)
string(LENGTH "${COLOURWARS_NEW_LEVEL_TEMPLATE}" colourwars_template_length)
string(FIND "${COLOURWARS_NEW_LEVEL_TEMPLATE}" ")template\"" colourwars_template_delimiter)
string(FIND "${COLOURWARS_NEW_LEVEL_TEMPLATE}" "@" colourwars_template_at)
if(colourwars_template_length GREATER 16000 OR NOT colourwars_template_delimiter EQUAL -1
    OR NOT colourwars_template_at EQUAL -1)
    message(FATAL_ERROR "editor/template.json cannot be compiled in as File > New: keep it under "
        "16000 bytes, without an @ or the text )template\".")
endif()
file(CONFIGURE OUTPUT "${PROJECT_BINARY_DIR}/generated/editor/new_level_template.h"
    CONTENT "#pragma once

// Generated by editor/CMakeLists.txt from editor/template.json; read by File > New.
namespace colourwars_editor
{
    inline constexpr char new_level_template[] = R\"template(${COLOURWARS_NEW_LEVEL_TEMPLATE})template\";
}
"
    @ONLY)
target_include_directories(colourwars_level_document PRIVATE "${PROJECT_BINARY_DIR}/generated")
